// Synthetic provider boundary, copied into disposable apps only. No fetch, no real credentials.
// Phase 1 and 2 behavior is unchanged; Phase 3 adds a stateful in-memory GitHub
// (branches, commits, pull requests, merges) persisted beside the scenario file
// so the web and worker processes see the same repository.
import { existsSync, readFileSync, writeFileSync } from "node:fs";
import { join } from "node:path";
import { gzipSync } from "node:zlib";
import { contentsResponse, fixtureCommitSha, fixtureFiles, treeResponse } from "./qa-fixtures/github-responses";
import { SyntheticGithub, SyntheticGithubError } from "./qa-fixtures/synthetic-github";
import { writeTar } from "./tar";

export class GithubAccessError extends Error {
  constructor(message: string, public readonly kind: "auth" | "access" | "unavailable" = "unavailable") { super(message); }
}
function scenario() {
  if (!process.env.SERVER_GUY_QA_ROOT) throw new Error("QA root required");
  try { return JSON.parse(readFileSync(join(process.env.SERVER_GUY_CONFIG_DIR!, "github-scenario.json"), "utf8")); }
  catch { return {}; }
}
function statePath(fullName: string) {
  return join(process.env.SERVER_GUY_CONFIG_DIR!, `github-state-${fullName.replace("/", "--")}.json`);
}
/** Loads the repository's synthetic state, applies pending scenario actions, runs the request, saves. */
function withGithub<T>(fullName: string, work: (github: SyntheticGithub) => T): T {
  const github = new SyntheticGithub(fullName);
  const path = statePath(fullName);
  if (existsSync(path)) github.restore(JSON.parse(readFileSync(path, "utf8")));
  const current = scenario();
  github.revision = typeof current.revision === "string" ? current.revision : "1";
  github.permissions = current.push === "denied" ? { pull: true, push: false, admin: false } : { pull: true, push: true, admin: false };
  // A merge the test "performed on GitHub": applied once, then remembered.
  const applied = existsSync(`${path}.merged`) ? (JSON.parse(readFileSync(`${path}.merged`, "utf8")) as number[]) : [];
  if (typeof current.mergePull === "number" && !applied.includes(current.mergePull)) {
    try {
      github.merge(current.mergePull, current.mergeMethod ?? "squash");
      writeFileSync(`${path}.merged`, JSON.stringify([...applied, current.mergePull]), { mode: 0o600 });
    } catch { /* not mergeable yet */ }
  }
  try {
    return work(github);
  } finally {
    writeFileSync(path, JSON.stringify(github.snapshot()), { mode: 0o600 });
  }
}
export async function readGithubCliCredential() {
  const current = scenario();
  if (current.credential === "missing") return null;
  return { token: current.credential === "changed" ? "QA-CHANGED" : "QA-GITHUB-TOKEN", source: "gh" as const };
}
export async function githubDeviceRequest(path: string, body: Record<string, string>) {
  const current = scenario();
  if (path === "/login/device/code") return { device_code: "QA-DEVICE-SECRET", user_code: "ABCD-1234", verification_uri: "https://github.com/login/device", expires_in: 900, interval: 1 };
  if (body.grant_type === "refresh_token") {
    if (current.refresh === "revoked") return { error: "bad_refresh_token", error_description: "QA-SECRET-PROVIDER-ERROR" };
    if (current.refresh === "offline") throw new GithubAccessError("Could not contact GitHub. Try again.");
    return { access_token: "ghu_QA-RENEWED", token_type: "bearer", expires_in: 28800, refresh_token: "ghr_QA-RENEWED", refresh_token_expires_in: 15897600 };
  }
  if (current.login === "pending") return { error: "authorization_pending" };
  if (current.login === "denied") return { error: "access_denied" };
  if (current.login === "expired") return { error: "expired_token" };
  return { access_token: "ghu_QA-SYNTHETIC", token_type: "bearer", expires_in: 28800, refresh_token: "ghr_QA-SYNTHETIC", refresh_token_expires_in: 15897600 };
}
export async function githubArchive(fullName: string, sha: string) {
  return withGithub(fullName, (github) => {
    try {
      return github.archive(sha, writeTar);
    } catch (error) {
      throw new GithubAccessError(error instanceof Error ? error.message : "archive failed", "access");
    }
  });
}
export async function githubJson(path: string, _token: string, options: { method?: "POST" | "PATCH"; body?: unknown; allowNotFound?: boolean } = {}) {
  const current = scenario();
  if (current.repository === "revoked") throw new GithubAccessError("GitHub no longer accepts this login. Reconnect in Settings → GitHub.", "auth");
  let data: unknown;
  if (path === "/user") data = { id: 42, login: "qa-fixture-user" };
  else if (path.startsWith("/user/installations?")) data = { installations: [{ id: 7, app_slug: "qa-server-guy", account: { id: 1, login: "qa" }, permissions: { contents: current.repository === "missing-scope" ? "none" : "read", metadata: "read" }, repository_selection: "selected", suspended_at: null }] };
  else if (path.startsWith("/user/installations/7/repositories")) data = { repositories: current.repository === "denied" ? [] : [{ id: 99 }] };
  else if (path.startsWith("/user/installations/")) data = { id: 7, permissions: { contents: "write", pull_requests: "write", metadata: "read" } };
  else {
    const [, , owner, name, action] = path.split("/");
    if (!owner || !name) throw new Error("Unknown fixture endpoint");
    if (name.includes("missing")) throw new GithubAccessError("The repository is missing or this login cannot access it.", "access");
    if (name.includes("offline")) throw new GithubAccessError("Could not reach GitHub.");
    if (name.includes("slow-create") && !action) await new Promise((resolve) => setTimeout(resolve, 3000));
    const fullName = `${owner}/${name}`;
    // Phase 2 reads are pinned to one synthetic commit per repository name and
    // scenario revision; `revision: "2"` in the scenario simulates a new push.
    const revision = typeof current.revision === "string" ? current.revision : "1";
    const route = action?.split("?")[0];
    if (current.inspection === "offline" && (action === "git" || action === "contents")) throw new GithubAccessError("Could not reach GitHub.");
    const stateful = ["git", "pulls", "compare", "contents", "commits"].includes(route ?? "") || options.method;
    if (action === "git" && path.includes("/git/trees/") && !options.method && !existsSync(statePath(fullName))) {
      const sha = path.split("/")[6]?.split("?")[0] ?? "";
      data = treeResponse(fullName, sha);
    } else if (stateful) {
      try {
        return await withGithub(fullName, (github) => github.request(path.replace(`/repos/${fullName}`, `/repos/${github["fullName"]}`), options));
      } catch (error) {
        if (error instanceof SyntheticGithubError) throw new GithubAccessError(error.message, error.kind);
        throw error;
      }
    } else if (action === "contents") {
      const [encoded, query = ""] = path.slice(path.indexOf("/contents/") + "/contents/".length).split("?");
      const body = contentsResponse(fullName, decodeURIComponent(encoded), new URLSearchParams(query).get("ref") ?? "");
      if (!body) throw new GithubAccessError("The repository is missing or this login cannot access it.", "access");
      data = body;
    } else if (route === "commits") data = { sha: fixtureCommitSha(fullName, revision) };
    else data = { id: 99, full_name: fullName, visibility: "private", default_branch: "main", permissions: { pull: true, push: current.push !== "denied", admin: false } };
  }
  return { data, scopes: ["repo"] };
}
