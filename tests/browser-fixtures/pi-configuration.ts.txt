// Copied only into disposable QA apps. The production Pi adapter, tools,
// native sessions and worker stay intact; only model output is synthetic.
import {
  createAssistantMessageEventStream,
  InMemoryCredentialStore,

  type Context,
  type Model,
  type Api,
  type SimpleStreamOptions,
  type ToolCall,
} from "@earendil-works/pi-ai";
import { createHash, randomUUID } from "node:crypto";
import { existsSync, writeFileSync } from "node:fs";
import { join } from "node:path";
import {
  readPiConfiguration,
  readPiCredential,
  type PiSdk,
} from "./pi-configuration-real";
import {
  buildContractProposal,
  CONTRACT_READ_ORDER,
  treePaths,
  type SeenRead,
} from "./qa-fixtures/contract-builder";

export * from "./pi-configuration-real";

const INSPECTION_REQUEST =
  "Inspect the repository and propose the Application Contract.";

/**
 * The scripted Phase 2 flow: read the saved inspection, read the fixture
 * files that matter, propose a contract built from exactly those reads, then
 * answer. "contract: correct <value>" revises health.path from the engineer's
 * message; "contract: invented" cites a read that never happened.
 */
function contractStep(
  user: string,
  context: Context,
  userIndex: number,
): { tool?: ToolCall; answer?: string } {
  const mode =
    user === INSPECTION_REQUEST
      ? "propose"
      : user.slice("contract:".length).trim();
  const results = context.messages
    .slice(userIndex + 1)
    .filter((entry) => entry.role === "toolResult")
    .map((entry) => ({
      name: entry.toolName,
      isError: entry.isError,
      body: textOf(entry),
    }));
  const runContext = context.messages
    .slice(0, userIndex + 1)
    .filter((entry) => entry.role === "user" && textOf(entry).startsWith("{"))
    .map((entry) => JSON.parse(textOf(entry)) as { userMessageId?: string })
    .at(-1);
  const call = (name: string, args: Record<string, unknown>): ToolCall => ({
    type: "toolCall",
    id: randomUUID(),
    name,
    arguments: args,
  });
  const inspectionResult = results.find(
    (result) => result.name === "get_repository_inspection" && !result.isError,
  );
  const proposalResult = results.find(
    (result) => result.name === "propose_application_contract",
  );
  const currentResult = results.find(
    (result) => result.name === "get_application_contract" && !result.isError,
  );
  if (!inspectionResult) return { tool: call("get_repository_inspection", {}) };
  if (proposalResult) {
    if (proposalResult.isError)
      return {
        answer: `[QA contract] Proposal rejected; nothing was saved. ${proposalResult.body.split("\n")[1] ?? ""}`,
      };
    const proposal = JSON.parse(proposalResult.body) as {
      version: number;
      fields: number;
      blockers: string[];
      conformanceItems: string[];
    };
    return {
      answer: `[QA contract] Proposed Application Contract v${proposal.version} with ${proposal.fields} fields, ${proposal.blockers.length} blocker(s) and ${proposal.conformanceItems.length} conformance item(s).`,
    };
  }
  const inspection = JSON.parse(inspectionResult.body) as {
    inspection: { observationId: string; commitSha: string } | null;
    tree: { entries: string[] };
  };
  if (!inspection.inspection)
    return {
      answer:
        "[QA contract] The repository has not been inspected in this phase.",
    };
  const paths = treePaths(inspection);
  const reads: SeenRead[] = results
    .filter((result) => result.name === "read_repository_file" && !result.isError)
    .map((result) => JSON.parse(result.body) as SeenRead);
  const next = CONTRACT_READ_ORDER.filter((path) => paths.includes(path)).find(
    (path) => !reads.some((read) => read.path === path),
  );
  if (next) return { tool: call("read_repository_file", { path: next }) };
  if (mode.startsWith("correct") && !currentResult)
    return { tool: call("get_application_contract", {}) };
  const current = currentResult
    ? (JSON.parse(currentResult.body) as { current: { id: string } | null })
        .current
    : null;
  const value = mode.split(" ").at(-1)!;
  return {
    tool: call(
      "propose_application_contract",
      buildContractProposal(inspection, reads, {
        ...(current ? { revises: current.id } : {}),
        ...(mode.startsWith("correct") && current
          ? {
              corrections: [
                {
                  key: "health.path",
                  value,
                  messageId: runContext?.userMessageId ?? "missing",
                  quote: value,
                },
              ],
            }
          : {}),
        invented: mode === "invented",
      }),
    ),
  };
}

function textOf(message: Context["messages"][number]) {
  return typeof message.content === "string"
    ? message.content
    : message.content
        .filter((part) => part.type === "text")
        .map((part) => part.text)
        .join("");
}

function pause(ms: number, signal?: AbortSignal) {
  return new Promise<void>((resolve, reject) => {
    const abort = () => {
      clearTimeout(timer);
      reject(new Error("Synthetic generation cancelled."));
    };
    const timer = setTimeout(() => {
      signal?.removeEventListener("abort", abort);
      resolve();
    }, ms);
    signal?.addEventListener("abort", abort, { once: true });
    if (signal?.aborted) abort();
  });
}

function streamSynthetic(
  model: Model<Api>,
  context: Context,
  options?: SimpleStreamOptions,
) {
  const stream = createAssistantMessageEventStream();
  const message: AssistantMessage = {
    role: "assistant",
    content: [],
    api: model.api,
    provider: model.provider,
    model: model.id,
    stopReason: "stop",
    timestamp: Date.now(),
    usage: {
      input: 10,
      output: 10,
      cacheRead: 0,
      cacheWrite: 0,
      totalTokens: 20,
      cost: { input: 0, output: 0, cacheRead: 0, cacheWrite: 0, total: 0 },
    },
  };
  void (async () => {
    try {
      const root = process.env.SERVER_GUY_QA_ROOT;
      if (!root)
        throw new Error("Synthetic provider requires an isolated QA root.");
      options?.signal?.throwIfAborted();
      const userIndex = context.messages.findLastIndex(
        (entry) => entry.role === "user",
      );
      const userMessage =
        userIndex < 0 ? "" : textOf(context.messages[userIndex]);
      const results = context.messages
        .slice(userIndex + 1)
        .filter((entry) => entry.role === "toolResult");
      const lastResult = results.at(-1);
      const searchResults = results
        .filter(
          (entry) => entry.toolName === "search_decisions" && !entry.isError,
        )
        .map(
          (entry) =>
            JSON.parse(textOf(entry)) as {
              records: Array<{ id: string; value: string }>;
              nextOffset: number | null;
            },
        );
      const lastSearch = searchResults.at(-1);
      let tool: ToolCall | undefined;
      let answer = `[QA fixture reply] ${userMessage}`;
      if (
        userMessage === INSPECTION_REQUEST ||
        userMessage.startsWith("contract:")
      ) {
        const step = contractStep(userMessage, context, userIndex);
        tool = step.tool;
        if (step.answer) answer = step.answer;
      } else if (lastResult?.isError) {
        answer =
          "[QA fixture reply] Replacement rejected; no Decision was staged.";
      } else if (
        userMessage.startsWith("replace-priority: ") ||
        userMessage === "lookup-decisions"
      ) {
        if (!lastSearch || lastSearch.nextOffset !== null) {
          tool = {
            type: "toolCall",
            id: randomUUID(),
            name: "search_decisions",
            arguments: { offset: lastSearch?.nextOffset ?? 0 },
          };
        } else if (userMessage === "lookup-decisions") {
          answer = `[QA saved Decisions] ${
            searchResults
              .flatMap((result) => result.records)
              .map((record) => record.value)
              .join("; ") || "none"
          }`;
        } else if (
          !results.some((entry) => entry.toolName === "propose_decision")
        ) {
          const previous = searchResults
            .flatMap((result) => result.records)
            .at(-1);
          tool = {
            type: "toolCall",
            id: randomUUID(),
            name: "propose_decision",
            arguments: {
              kind: "launch-priority",
              value: userMessage.slice(18),
              ...(previous ? { replaces: previous.id } : {}),
            },
          };
        }
      } else if (
        !results.length &&
        (userMessage.startsWith("priority: ") ||
          userMessage.startsWith("invalid-replacement: "))
      ) {
        tool = {
          type: "toolCall",
          id: randomUUID(),
          name: "propose_decision",
          arguments: {
            kind: "launch-priority",
            value: userMessage.startsWith("priority: ")
              ? userMessage.slice(10)
              : "Must not persist",
            ...(userMessage.startsWith("invalid-replacement: ")
              ? { replaces: "00000000-0000-4000-8000-000000000000" }
              : {}),
          },
        };
      } else if (userMessage.startsWith("recall: ")) {
        const phrase = userMessage.slice(8);
        const found = context.messages
          .slice(0, userIndex)
          .some((entry) => entry.role === "user" && textOf(entry) === phrase);
        answer = `[QA native history] ${found ? "found" : "absent"}: ${phrase}`;
      }

      stream.push({ type: "start", partial: message });
      if (tool) {
        message.content = [tool];
        message.stopReason = "toolUse";
        stream.push({
          type: "toolcall_start",
          contentIndex: 0,
          partial: message,
        });
        stream.push({
          type: "toolcall_end",
          contentIndex: 0,
          toolCall: tool,
          partial: message,
        });
        stream.push({ type: "done", reason: "toolUse", message });
        return;
      }

      // Stream a genuine prefix of the final response; interrupted drafts are
      // saved by the production worker and retained in the native transcript.
      const prefix = answer.startsWith("[QA fixture reply]")
        ? "[QA fixture reply] "
        : answer;
      message.content = [{ type: "text", text: prefix }];
      stream.push({ type: "text_start", contentIndex: 0, partial: message });
      stream.push({
        type: "text_delta",
        contentIndex: 0,
        delta: prefix,
        partial: message,
      });
      // Deterministic crash barrier: stay after the persisted draft until the
      // process is killed or cancellation settles normally. Match only a prefix
      // so a later "recall: ..." request can quote this marker without pausing.
      if (userMessage.startsWith("[pause-after-draft]")) {
        const signal = options?.signal;
        if (!signal)
          throw new Error("The draft barrier requires cancellation.");
        await new Promise<never>((_resolve, reject) => {
          const abort = () => {
            signal.removeEventListener("abort", abort);
            reject(new Error("Synthetic generation cancelled."));
          };
          signal.addEventListener("abort", abort, { once: true });
          if (signal.aborted) abort();
        });
      }
      // The process-lifecycle regression deliberately models a provider that
      // never settles, even after cancellation. Used only in a disposable child.
      if (userMessage.includes("[unresponsive]")) {
        await new Promise<void>(() => {});
      }
      await pause(
        userMessage.includes("[slow-cancel]")
          ? 6_000
          : userMessage.includes("[slow]")
            ? 1_500
            : 150,
        options?.signal,
      );
      const failureMarker = join(
        root,
        `pi-failed-${createHash("sha256").update(userMessage).digest("hex")}`,
      );
      if (userMessage.includes("[fail-once]") && !existsSync(failureMarker)) {
        writeFileSync(failureMarker, "failed", { mode: 0o600 });
        throw new Error("QA simulated provider failure.");
      }
      message.content = [{ type: "text", text: answer }];
      stream.push({
        type: "text_delta",
        contentIndex: 0,
        delta: answer.slice(prefix.length),
        partial: message,
      });
      stream.push({
        type: "text_end",
        contentIndex: 0,
        content: answer,
        partial: message,
      });
      stream.push({ type: "done", reason: "stop", message });
    } catch (error) {
      message.stopReason = options?.signal?.aborted ? "aborted" : "error";
      message.errorMessage =
        error instanceof Error ? error.message : "Synthetic provider failed.";
      stream.push({
        type: "error",
        reason: message.stopReason,
        error: message,
      });
    }
  })();
  return stream;
}

export async function configuredPiRuntime(sdk: PiSdk) {
  const root = process.env.SERVER_GUY_QA_ROOT;
  if (!root)
    throw new Error("Synthetic provider requires an isolated QA root.");
  const configuration = readPiConfiguration();
  if (
    !configuration ||
    !readPiCredential(configuration.authPath, configuration.providerId)
  )
    throw new Error("Connect ChatGPT in Settings before using the QA model.");
  const modelRuntime = await sdk.ModelRuntime.create({
    credentials: new InMemoryCredentialStore(),
    modelsPath: null,
    modelsStorePath: join(root, "synthetic-models.json"),
    refreshOnCreate: false,
    allowModelNetwork: false,
  });
  // A separate provider ID and synthetic credential ensure the runtime cannot
  // fall back to a real account or a network provider adapter.
  modelRuntime.registerProvider("server-guy-qa", {
    api: "server-guy-qa",
    apiKey: "QA-NO-NETWORK",
    baseUrl: "https://qa.invalid",
    streamSimple: streamSynthetic,
    models: [
      {
        id: configuration.modelId,
        name: "Synthetic QA model",
        reasoning: true,
        input: ["text"],
        contextWindow: 200_000,
        maxTokens: 8_000,
        cost: { input: 0, output: 0, cacheRead: 0, cacheWrite: 0 },
      },
    ],
  });
  const model = modelRuntime.getModel("server-guy-qa", configuration.modelId)!;
  return { configuration, modelRuntime, model };
}
