// Copied only into disposable QA apps. The production Pi adapter, tools,
// native sessions and worker stay intact; only model output is synthetic.
import {
  createAssistantMessageEventStream,
  InMemoryCredentialStore,
  type AssistantMessage,
  type Context,
  type Model,
  type Api,
  type SimpleStreamOptions,
} from "@earendil-works/pi-ai";
import { createHash } from "node:crypto";
import { existsSync, writeFileSync } from "node:fs";
import { join } from "node:path";
import {
  readPiConfiguration,
  readPiCredential,
  type PiSdk,
} from "./pi-configuration-real";

export * from "./pi-configuration-real";

function textOf(message: Context["messages"][number]) {
  return typeof message.content === "string"
    ? message.content
    : message.content
        .filter((part) => part.type === "text")
        .map((part) => part.text)
        .join("");
}

function pause(ms: number, signal?: AbortSignal) {
  return new Promise<void>((resolve, reject) => {
    const abort = () => {
      clearTimeout(timer);
      reject(new Error("Synthetic generation cancelled."));
    };
    const timer = setTimeout(() => {
      signal?.removeEventListener("abort", abort);
      resolve();
    }, ms);
    signal?.addEventListener("abort", abort, { once: true });
    if (signal?.aborted) abort();
  });
}

function streamSynthetic(
  model: Model<Api>,
  context: Context,
  options?: SimpleStreamOptions,
) {
  const stream = createAssistantMessageEventStream();
  const message: AssistantMessage = {
    role: "assistant",
    content: [],
    api: model.api,
    provider: model.provider,
    model: model.id,
    stopReason: "stop",
    timestamp: Date.now(),
    usage: {
      input: 10,
      output: 10,
      cacheRead: 0,
      cacheWrite: 0,
      totalTokens: 20,
      cost: { input: 0, output: 0, cacheRead: 0, cacheWrite: 0, total: 0 },
    },
  };
  void (async () => {
    try {
      const root = process.env.HALLVI_QA_ROOT;
      if (!root)
        throw new Error("Synthetic provider requires an isolated QA root.");
      options?.signal?.throwIfAborted();
      const userIndex = context.messages.findLastIndex(
        (entry) => entry.role === "user",
      );
      const userMessage =
        userIndex < 0 ? "" : textOf(context.messages[userIndex]);
      const answer = `[QA fixture reply] ${userMessage}`;

      stream.push({ type: "start", partial: message });
      // Stream a genuine prefix of the final response; interrupted drafts are
      // saved by the production worker and retained in the native transcript.
      const prefix = "[QA fixture reply] ";
      message.content = [{ type: "text", text: prefix }];
      stream.push({ type: "text_start", contentIndex: 0, partial: message });
      stream.push({
        type: "text_delta",
        contentIndex: 0,
        delta: prefix,
        partial: message,
      });
      await pause(
        userMessage.includes("[hold]")
          ? 60_000
          : userMessage.includes("[slow-cancel]")
          ? 6_000
          : userMessage.includes("[slow]")
            ? 1_500
            : 150,
        options?.signal,
      );
      const failureMarker = join(
        root,
        `pi-failed-${createHash("sha256").update(userMessage).digest("hex")}`,
      );
      if (userMessage.includes("[fail-once]") && !existsSync(failureMarker)) {
        writeFileSync(failureMarker, "failed", { mode: 0o600 });
        throw new Error("QA simulated provider failure.");
      }
      message.content = [{ type: "text", text: answer }];
      stream.push({
        type: "text_delta",
        contentIndex: 0,
        delta: answer.slice(prefix.length),
        partial: message,
      });
      stream.push({
        type: "text_end",
        contentIndex: 0,
        content: answer,
        partial: message,
      });
      stream.push({ type: "done", reason: "stop", message });
    } catch (error) {
      message.stopReason = options?.signal?.aborted ? "aborted" : "error";
      message.errorMessage =
        error instanceof Error ? error.message : "Synthetic provider failed.";
      stream.push({
        type: "error",
        reason: message.stopReason,
        error: message,
      });
    }
  })();
  return stream;
}

export async function configuredPiRuntime(sdk: PiSdk) {
  const root = process.env.HALLVI_QA_ROOT;
  if (!root)
    throw new Error("Synthetic provider requires an isolated QA root.");
  const configuration = readPiConfiguration();
  if (
    !configuration ||
    !readPiCredential(configuration.authPath, configuration.providerId)
  )
    throw new Error("Connect ChatGPT in Settings before using the QA model.");
  const modelRuntime = await sdk.ModelRuntime.create({
    credentials: new InMemoryCredentialStore(),
    modelsPath: null,
    modelsStorePath: join(root, "synthetic-models.json"),
    refreshOnCreate: false,
    allowModelNetwork: false,
  });
  // A separate provider ID and synthetic credential ensure the runtime cannot
  // fall back to a real account or a network provider adapter.
  modelRuntime.registerProvider("hallvi-qa", {
    api: "hallvi-qa",
    apiKey: "QA-NO-NETWORK",
    baseUrl: "https://qa.invalid",
    streamSimple: streamSynthetic,
    models: [
      {
        id: configuration.modelId,
        name: "Synthetic QA model",
        reasoning: true,
        input: ["text"],
        contextWindow: 200_000,
        maxTokens: 8_000,
        cost: { input: 0, output: 0, cacheRead: 0, cacheWrite: 0 },
      },
    ],
  });
  const model = modelRuntime.getModel("hallvi-qa", configuration.modelId)!;
  return { configuration, modelRuntime, model };
}
