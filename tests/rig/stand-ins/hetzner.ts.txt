// Rig stand-in for the Hetzner Cloud API, copied into the rig's app copy
// only. It prices one fixed small x86 offer (the value the existing Docker
// proof harness uses) and "creates" this machine at 127.0.0.1, reached through
// the rig's local SSH stand-in. State persists in <config>/rig-cloud.json so
// the web and worker processes agree. Request bodies are not logged: server
// creation carries the host's private SSH key in user_data.
import {
  appendFileSync,
  existsSync,
  readFileSync,
  renameSync,
  writeFileSync,
} from "node:fs";
import { join } from "node:path";
import type { HostOffer } from "./deployment-types";

type Labelled = { id: number; labels?: Record<string, string> } & Record<
  string,
  unknown
>;
type Cloud = {
  next: number;
  servers: Labelled[];
  sshKeys: Labelled[];
  firewalls: Labelled[];
};

const file = () => join(process.env.SERVER_GUY_CONFIG_DIR!, "rig-cloud.json");
function load(): Cloud {
  return existsSync(file())
    ? JSON.parse(readFileSync(file(), "utf8"))
    : { next: 4200, servers: [], sshKeys: [], firewalls: [] };
}
function save(cloud: Cloud) {
  const temporary = `${file()}.tmp`;
  writeFileSync(temporary, JSON.stringify(cloud, null, 2));
  renameSync(temporary, file());
}
function log(entry: Record<string, unknown>) {
  appendFileSync(
    join(process.env.SERVER_GUY_CONFIG_DIR!, "rig-cloud.log"),
    JSON.stringify({ at: new Date().toISOString(), ...entry }) + "\n",
  );
}

export function hetznerConnectionId() {
  return "audit-local-provider";
}
// Only documented rejection outcomes prove that a create request had no effect.
export class HetznerError extends Error {
  constructor(
    public readonly status: number,
    public readonly code: string,
  ) {
    super(
      `Hetzner rejected the request (HTTP ${status}, ${code}). Check access, capacity and the recorded operation before retrying.`,
    );
  }
  get definitelyNotCreated() {
    return this.status >= 400 && this.status < 500;
  }
}

function labelled(items: Labelled[], search: URLSearchParams) {
  const selector = search.get("label_selector");
  if (!selector) return items;
  const [key, value] = selector.split("=");
  return items.filter((item) => item.labels?.[key] === value);
}
export async function hetzner<T>(
  path: string,
  body?: unknown,
  _token?: string,
  // The product asks for a method explicitly — a POST with no body, a DELETE
  // — so the stand-in has to take one rather than guess from the body.
  method = body === undefined ? "GET" : "POST",
  _signal?: AbortSignal,
): Promise<T> {
  const url = new URL(path, "https://provider.invalid");
  const input = (body ?? {}) as Record<string, unknown>;
  log({ method, path, bodyKeys: Object.keys(input) });
  const cloud = load();
  const route = url.pathname;
  const byId = (items: Labelled[]) => {
    const item = items.find((x) => x.id === Number(route.split("/")[2]));
    if (!item) throw new HetznerError(404, "not_found");
    return item;
  };
  let result: unknown;
  if (route === "/servers" && method === "POST") {
    const server: Labelled = {
      id: cloud.next++,
      name: input.name,
      status: "running",
      labels: input.labels as Record<string, string>,
      server_type: { name: input.server_type },
      datacenter: { location: { name: input.location } },
      public_net: {
        // TEST-NET-1 (RFC 5737): reserved for documentation and not routable,
        // so nothing can leave the machine, while still being something the
        // product and the model can treat as a public address. bin/ssh never
        // reads it — every host command runs locally or in the host
        // container — so this is the fiction's address, not a destination.
        ipv4: { ip: "192.0.2.10" },
        firewalls: ((input.firewalls ?? []) as { firewall: number }[]).map(
          (item) => ({ id: item.firewall, status: "applied" }),
        ),
      },
    };
    cloud.servers.push(server);
    result = { server };
  } else if (route === "/servers")
    result = { servers: labelled(cloud.servers, url.searchParams) };
  else if (/^\/servers\/\d+$/.test(route))
    result = { server: byId(cloud.servers) };
  else if (route === "/ssh_keys" && method === "POST") {
    const key: Labelled = {
      id: cloud.next++,
      name: input.name,
      labels: input.labels as Record<string, string>,
    };
    cloud.sshKeys.push(key);
    // A real provider installs the key it is given; without that the product
    // authenticates against nothing. Rig B has a real sshd, so the key goes
    // where sshd will read it — which is what makes a fresh application work
    // without anybody authorizing it by hand.
    const container = process.env.SG_RIG_HOST_CONTAINER;
    const material =
      typeof input.public_key === "string" ? input.public_key : "";
    if (container && material.startsWith("ssh-"))
      try {
        const { execFileSync } = await import("node:child_process");
        execFileSync("docker", [
          "exec",
          container,
          "sh",
          "-c",
          `set -e
mkdir -p /root/.ssh
chmod 700 /root/.ssh
touch /root/.ssh/authorized_keys
grep -qxF -- "$1" /root/.ssh/authorized_keys || printf '%s\\n' "$1" >> /root/.ssh/authorized_keys
chmod 600 /root/.ssh/authorized_keys`,
          "rig-authorize",
          material,
        ]);
      } catch {
        // The proof will fail at the connection instead, which is louder.
      }
    result = { ssh_key: key };
  } else if (route === "/ssh_keys")
    result = { ssh_keys: labelled(cloud.sshKeys, url.searchParams) };
  else if (route === "/firewalls" && method === "POST") {
    const firewall: Labelled = {
      id: cloud.next++,
      name: input.name,
      labels: input.labels as Record<string, string>,
      rules: input.rules ?? [],
    };
    cloud.firewalls.push(firewall);
    result = { firewall };
  } else if (route === "/firewalls")
    result = { firewalls: labelled(cloud.firewalls, url.searchParams) };
  else if (/^\/firewalls\/\d+$/.test(route))
    result = { firewall: byId(cloud.firewalls) };
  else if (/^\/firewalls\/\d+\/actions\/set_rules$/.test(route)) {
    byId(cloud.firewalls).rules = input.rules;
    result = { actions: [{ id: cloud.next++, status: "success" }] };
  } else if (/^\/actions\/\d+$/.test(route))
    result = { action: { id: Number(route.split("/")[2]), status: "success" } };
  else if (route === "/server_types")
    // Pi reads the catalog before it recommends a size. Without these the
    // only source of truth was /pricing, which it had to work around.
    result = {
      server_types: [
        {
          id: 1,
          name: "cx23",
          description: "CX23",
          cores: 2,
          memory: 4,
          disk: 40,
          cpu_type: "shared",
          architecture: "x86",
          deprecated: false,
        },
      ],
    };
  else if (route === "/locations" || route === "/datacenters") {
    const location = {
      id: 1,
      name: "hel1",
      description: "Helsinki DC Park 1",
      country: "FI",
      city: "Helsinki",
      network_zone: "eu-central",
    };
    result =
      route === "/locations"
        ? { locations: [location] }
        : {
            datacenters: [
              {
                id: 1,
                name: "hel1-dc2",
                description: "Helsinki 1 DC 2",
                location,
                server_types: { supported: [1], available: [1] },
              },
            ],
          };
  } else if (route === "/images")
    result = {
      images: [
        {
          id: 1,
          name: "ubuntu-24.04",
          description: "Ubuntu 24.04",
          os_flavor: "ubuntu",
          architecture: "x86",
          status: "available",
        },
      ],
    };
  else if (route === "/pricing")
    // A fixed, clearly rig-shaped price. The product records cost as
    // `reported` by the provider, so a stand-in answer is the honest kind of
    // input for it: nothing here was measured.
    result = {
      pricing: {
        currency: "EUR",
        vat_rate: "0.000",
        primary_ips: [
          {
            type: "ipv4",
            prices: [
              {
                location: "hel1",
                price_monthly: { net: "0.5000000000", gross: "0.5000000000" },
                price_hourly: { net: "0.0008000000", gross: "0.0008000000" },
              },
            ],
          },
        ],
        server_types: [
          {
            id: 1,
            name: "cx23",
            prices: [
              {
                location: "hel1",
                price_monthly: { net: "5.4900000000", gross: "5.4900000000" },
                price_hourly: { net: "0.0089000000", gross: "0.0089000000" },
                included_traffic: 21990232555520,
                price_per_tb_traffic: {
                  net: "1.0000000000",
                  gross: "1.0000000000",
                },
              },
            ],
          },
        ],
      },
    };
  else throw new Error(`Unexpected provider request ${path}`);
  save(cloud);
  return result as T;
}
